# Market Alert Android App Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Kotlin Android app skeleton that shows six market indicators in order, refreshes them from replaceable public/mock providers, persists latest quotes, and posts scheduled market notifications.

**Architecture:** Use one Android app module with clear packages for `domain`, `data`, `scheduler`, `worker`, `notifications`, and `ui`. Alarms trigger schedule slots, a WorkManager worker performs fetch/persist/notify work, and Compose renders the latest repository state. Public endpoints are isolated behind provider interfaces so brittle web sources can be replaced without touching UI or scheduling.

**Tech Stack:** Kotlin, Android Gradle Plugin, Jetpack Compose Material 3, Room, WorkManager, Retrofit, OkHttp, kotlinx.serialization, coroutines, JUnit.

## Global Constraints

- All schedules are based on Asia/Seoul local time.
- The fixed indicator order is USD/KRW, CNN Fear & Greed, S&P 500, Nasdaq 100, KRX gold spot 1g, KOSPI.
- Each indicator displays current value, absolute change, percent change when available, last successful update time, data state, and data source label.
- Use inexact `AlarmManager` alarms by default; do not implement exact alarm permission flow in the initial skeleton.
- Android 13 and later require `POST_NOTIFICATIONS` runtime permission before posting notifications.
- Use Room for local quote history and fetch logs.
- Use Retrofit, OkHttp, and kotlinx.serialization for HTTP JSON parsing.
- Public source parsing must be defensive and isolated behind providers.
- If every value in a slot fails, send no notification and update in-app state.
- Quotes are `FRESH` for 90 minutes after observation, `STALE` after 90 minutes when a prior value exists, and `UNAVAILABLE` when no prior value exists.
- A partial-success scheduled refresh must not return `Result.retry()` because retrying would risk duplicate notifications for the successful indicators; only all-failure retryable slots return `Result.retry()`.
- Do not add user-editable schedules, watchlists, login, paid APIs, charts, or historical analytics in the initial skeleton.

```kotlin
package com.nevermiss.opportunity.domain

import java.time.LocalTime
import java.time.ZoneId

data class ScheduleSlot(
    val id: String,
    val time: LocalTime,
    val indicators: List<Indicator>,
)

object MarketSchedule {
    val zoneId: ZoneId = ZoneId.of("Asia/Seoul")

    val dailySlots: List<ScheduleSlot> = buildList {
        for (hour in 9..21) {
            addSlot(LocalTime.of(hour, 0), Indicator.USD_KRW)
        }
        addSlot(LocalTime.of(9, 0), Indicator.CNN_FEAR_GREED)
        addSlot(LocalTime.of(23, 0), Indicator.CNN_FEAR_GREED)
        listOf(LocalTime.of(22, 40), LocalTime.of(23, 0), LocalTime.of(23, 40)).forEach { time ->
            addSlot(time, Indicator.SP500)
            addSlot(time, Indicator.NASDAQ100)
        }
        addSlot(LocalTime.of(12, 0), Indicator.KRX_GOLD_1G)
        for (hour in 9..15) {
            addSlot(LocalTime.of(hour, 0), Indicator.KOSPI)
        }
    }
        .groupBy { it.time }
        .map { (time, slots) ->
            ScheduleSlot(
                id = "%02d%02d".format(time.hour, time.minute),
                time = time,
                indicators = Indicator.displayOrder.filter { indicator ->
                    slots.any { indicator in it.indicators }
                },
            )
        }
        .sortedWith(compareBy<ScheduleSlot> { it.time.hour }.thenBy { it.time.minute })

    fun slotsFor(indicator: Indicator): List<ScheduleSlot> =
        dailySlots.filter { indicator in it.indicators }

    fun slotAt(time: LocalTime): ScheduleSlot =
        dailySlots.first { it.time == time }

    private fun MutableList<ScheduleSlot>.addSlot(time: LocalTime, indicator: Indicator) {
        add(ScheduleSlot(id = "%02d%02d-%s".format(time.hour, time.minute, indicator.name), time = time, indicators = listOf(indicator)))
    }
}
```

```kotlin
package com.nevermiss.opportunity.data

import com.nevermiss.opportunity.data.local.FetchLogEntity
import com.nevermiss.opportunity.data.local.MarketDao
import com.nevermiss.opportunity.data.local.toDomain
import com.nevermiss.opportunity.data.local.toEntity
import com.nevermiss.opportunity.domain.Indicator
import com.nevermiss.opportunity.domain.MarketQuote
import com.nevermiss.opportunity.domain.QuoteState
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.map

class DefaultMarketRepository(
    private val dao: MarketDao,
    private val provider: MarketDataProvider,
    private val clockMillis: () -> Long = System::currentTimeMillis,
) : MarketRepository {
    override fun observeLatestQuotes(): Flow<Map<Indicator, MarketQuote>> =
        dao.observeLatestQuotes().map { rows ->
            rows.associate {
                val quote = it.toDomain()
                it.indicator to quote.copy(state = QuoteFreshness.stateFor(quote, clockMillis()))
            }
        }

    override suspend fun latestQuotes(): Map<Indicator, MarketQuote> =
        dao.latestQuotes().associate {
            val quote = it.toDomain()
            it.indicator to quote.copy(state = QuoteFreshness.stateFor(quote, clockMillis()))
        }

    override suspend fun refreshIndicators(indicators: List<Indicator>, slotId: String): RefreshResult {
        val startedAt = clockMillis()
        val successes = mutableListOf<MarketQuote>()
        val failures = mutableListOf<ProviderResult.Failure>()
        val quotesBeforeRefresh = latestQuotes()

        indicators.forEach { indicator ->
            when (val result = provider.fetch(indicator)) {
                is ProviderResult.Success -> {
                    successes += result.quote
                    dao.insertQuote(result.quote.toEntity())
                }
                is ProviderResult.Failure -> {
                    failures += result
                    quotesBeforeRefresh[indicator]
                        ?.copy(state = QuoteState.STALE)
                        ?.let { dao.insertQuote(it.toEntity()) }
                }
            }
        }

        dao.insertFetchLog(
            FetchLogEntity(
                slotId = slotId,
                startedAtEpochMillis = startedAt,
                completedAtEpochMillis = clockMillis(),
                successCount = successes.size,
                failureCount = failures.size,
                errorSummary = failures.joinToString { "${it.indicator.name}:${it.reason}" },
            ),
        )

        return RefreshResult(slotId, successes, failures)
    }
}
```

```kotlin
import retrofit2.http.GET

class YahooFinanceProvider(
    baseUrl: String = "https://query1.finance.yahoo.com/",
    client: OkHttpClient = OkHttpClient(),
    private val clockMillis: () -> Long = System::currentTimeMillis,
) : MarketDataProvider {
    private val json = Json { ignoreUnknownKeys = true }
    private val api = Retrofit.Builder()
        .baseUrl(baseUrl)
        .client(client)
        .addConverterFactory(json.asConverterFactory("application/json".toMediaType()))
        .build()
        .create(YahooApi::class.java)

    override suspend fun fetch(indicator: Indicator): ProviderResult {
        val symbol = indicator.yahooSymbol
            ?: return ProviderResult.Failure(indicator, "missing-yahoo-symbol", retryable = false)

        return runCatching {
            val response = api.chart(symbol)
            val meta = response.chart.result?.firstOrNull()?.meta
                ?: return ProviderResult.Failure(indicator, "empty-chart", retryable = true)
            val value = meta.regularMarketPrice
                ?: return ProviderResult.Failure(indicator, "missing-price", retryable = true)
            val previous = meta.previousClose ?: meta.chartPreviousClose ?: value
            val change = value - previous
            val percent = if (previous == 0.0) null else (change / previous) * 100.0
            ProviderResult.Success(
                MarketQuote(
                    indicator = indicator,
                    value = value,
                    change = change,
                    percentChange = percent,
                    observedAtEpochMillis = (meta.regularMarketTime?.times(1000)) ?: clockMillis(),
                    source = indicator.sourceLabel,
                    state = QuoteState.FRESH,
                ),
            )
        }.getOrElse {
            ProviderResult.Failure(indicator, it.javaClass.simpleName, retryable = true)
        }
    }
}
```

```kotlin
    fun canPostNotifications(): Boolean =
```

```kotlin
package com.nevermiss.opportunity.scheduler

import android.app.AlarmManager
import android.app.PendingIntent
import android.content.Context
import android.content.Intent
import com.nevermiss.opportunity.domain.MarketSchedule
import java.time.ZonedDateTime

class MarketAlarmScheduler(private val context: Context) {
    private val alarmManager = context.getSystemService(AlarmManager::class.java)

    fun scheduleAll(now: ZonedDateTime = ZonedDateTime.now(MarketSchedule.zoneId)) {
        MarketSchedule.dailySlots.forEach { slot ->
            val triggerAt = nextOccurrence(now, slot.time).toInstant().toEpochMilli()
            alarmManager.set(
                AlarmManager.RTC_WAKEUP,
                triggerAt,
                pendingIntent(slot.id),
            )
        }
    }

    private fun pendingIntent(slotId: String): PendingIntent {
        val intent = Intent(context, MarketAlarmReceiver::class.java).putExtra(MarketAlarmReceiver.EXTRA_SLOT_ID, slotId)
        return PendingIntent.getBroadcast(context, slotId.hashCode(), intent, PendingIntent.FLAG_IMMUTABLE or PendingIntent.FLAG_UPDATE_CURRENT)
    }

    companion object {
        fun nextOccurrence(now: ZonedDateTime, time: java.time.LocalTime): ZonedDateTime {
            val today = now.toLocalDate().atTime(time).atZone(MarketSchedule.zoneId)
            return if (today.isAfter(now)) today else today.plusDays(1)
        }
    }
}
```

```kotlin
package com.nevermiss.opportunity.worker

import android.content.Context
import androidx.work.CoroutineWorker
import androidx.work.WorkerParameters
import com.nevermiss.opportunity.AppGraph
import com.nevermiss.opportunity.data.RefreshResult
import com.nevermiss.opportunity.domain.MarketSchedule

class MarketFetchWorker(
    context: Context,
    params: WorkerParameters,
) : CoroutineWorker(context, params) {
    override suspend fun doWork(): Result {
        val slotId = inputData.getString(KEY_SLOT_ID) ?: return Result.failure()
        val slot = MarketSchedule.dailySlots.firstOrNull { it.id == slotId } ?: return Result.failure()
        val graph = AppGraph.get(applicationContext)
        val refresh = graph.repository.refreshIndicators(slot.indicators, slot.id)
        if (shouldPostNotification(refresh)) {
            graph.notifications.showMarketUpdate(refresh)
        }
        graph.scheduler.scheduleAll()
        // Partial successes are treated as success to avoid duplicate notifications on retry.
        return if (refresh.failures.any { it.retryable } && refresh.successes.isEmpty()) Result.retry() else Result.success()
    }

    companion object {
        const val KEY_SLOT_ID = "slot_id"

        fun shouldPostNotification(result: RefreshResult): Boolean = result.successes.isNotEmpty()
    }
}
```
